function doGet(e) {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('ASESMEN HARIAN PAI SDN RAWA BUAYA 01')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// Inisialisasi Lembar Spreadsheet
function setupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Sheet Hasil Ujian
  var sheetHasil = ss.getSheetByName("Hasil_Ujian") || ss.insertSheet("Hasil_Ujian");
  if (sheetHasil.getLastRow() === 0) {
    sheetHasil.appendRow(["Timestamp", "Nama Siswa", "Kelas", "Nilai PG", "Nilai PGK", "Nilai BS", "Nilai Isian", "Nilai Uraian", "Total Nilai", "Status"]);
  }
  
  // Sheet Data Siswa
  var sheetSiswa = ss.getSheetByName("Data_Siswa") || ss.insertSheet("Data_Siswa");
  if (sheetSiswa.getLastRow() === 0) {
    sheetSiswa.appendRow(["ID", "Nama Siswa", "Kelas"]);
  }
  
  // Sheet Pengaturan Token
  var sheetConfig = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
  if (sheetConfig.getLastRow() === 0) {
    sheetConfig.appendRow(["Key", "Value"]);
    sheetConfig.appendRow(["TOKEN_UJIAN", "PAI2026"]);
    sheetConfig.appendRow(["KATA_KUNCI_GURU", "120695Jrs"]);
  }
  
  return "Database Berhasil Diinisialisasi!";
}

// Simpan Hasil Ujian Siswa
function simpanHasilUjian(data) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Hasil_Ujian") || ss.insertSheet("Hasil_Ujian");
    sheet.appendRow([
      new Date(),
      data.nama,
      data.kelas,
      data.nilaiPG,
      data.nilaiPGK,
      data.nilaiBS,
      data.nilaiIsian,
      data.nilaiUraian,
      data.totalNilai,
      data.status
    ]);
    return { success: true, message: "Hasil ujian berhasil tersimpan di Google Sheet!" };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}
